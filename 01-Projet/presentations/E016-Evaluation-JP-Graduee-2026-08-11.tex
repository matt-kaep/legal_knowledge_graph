\begin{frame}[t]{Hit@K exact ou pertinence juridique graduée ?}
  \scriptsize
  \keybox{\small\textbf{Périmètre commun :} top--10 G7 par question $\cdot$ 7 540 positions $\cdot$ 7 487 couples question--JP uniques}

  \vspace{0em}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.47\textwidth}
      \begin{center}
        {\color{accent}\textbf{MÉTRIQUE ACTUELLE}}\\[0.05em]
        \large\textbf{Hit@K exact}
      \end{center}

      \vspace{0em}
      \textbf{Entrée}
      \begin{itemize}\setlength{\itemsep}{0.2em}
        \item les JP attendues dans le benchmark ;
        \item les $K$ premières JP retournées par G7.
      \end{itemize}

      \textbf{Mesure}
      \[
        \mathrm{Hit@K}(q)=
        \frac{|A_q\cap R_q[{:}K]|}{\min(|A_q|,K)}
      \]

      \colorbox{yellow!18}{\parbox{0.92\linewidth}{\centering
        Une JP utile mais absente des réponses attendues vaut \textbf{0}.}}
    \end{column}

    \begin{column}{0.47\textwidth}
      \begin{center}
        {\color{ok}\textbf{NOUVELLE MÉTHODE}}\\[0.05em]
        \large\textbf{Score gradué@K}
      \end{center}

      \vspace{0em}
      \textbf{Entrée du juge}
      \begin{itemize}\setlength{\itemsep}{0.2em}
        \item la question juridique ;
        \item une analyse détaillée de la jurisprudence retournée.
      \end{itemize}

      \textbf{Jugement et score}
      \begin{center}
        A = 1 $\cdot$ B = 0,5 $\cdot$ C/D/E/NJ = 0\\[0em]
        $\displaystyle \mathrm{Score@K}(q)=\frac{\sum_{i=1}^{K}\mathrm{gain}_i}{K}$
      \end{center}

      \colorbox{green!12}{\parbox{0.92\linewidth}{\centering
        Une autre JP peut rapporter des points si elle répond juridiquement à la question.}}
    \end{column}
  \end{columns}

  \vspace{-0.35em}
  \warnbox{\small\textbf{Diagnostic :} Hit bas + gradué bas $\Rightarrow$ JP faibles ; Hit bas + gradué haut $\Rightarrow$ réponses attendues incomplètes.}
  \status{E016 exploratoire sur \texttt{eval\_rich\_retrievable\_strict}, déjà consulté ; aucun tuning G7 autorisé.}
\end{frame}
