\section{Chantier 1 -- Panel métriques complet}

\begin{frame}{Chantier 1 -- pourquoi maintenant}
  \footnotesize
  \textbf{Problème} : M1 et M2 \emph{seuls} ne couvrent pas tous les angles de qualité d'un retrieval,
  et dégénèrent sur le régime $|\mathrm{GT}|=1$ qui représente la majorité de notre cohorte.

  \vspace{0.6em}
  \renewcommand{\arraystretch}{1.25}
  \setlength{\tabcolsep}{6pt}
  \begin{center}
  \begin{tabular}{@{}l p{5.5cm} p{4.5cm}@{}}
    \toprule
    \textbf{Métrique} & \textbf{Ce qu'elle évalue} & \textbf{Régime cible} \\
    \midrule
    M1 -- Recall@K & Couverture brute de la GT & multi-GT (étendu) \\
    M2 -- Rang moyen normalisé & Qualité du re-rank sur GT trouvée & multi-GT \\
    \rowcolor{green!10}
    \textbf{Hit@K} & $\ge 1$ GT dans le top-$K$ (binaire) & \textbf{singleton} (strict) \\
    \rowcolor{green!10}
    \textbf{MRR@K} & Rang du \emph{premier} GT trouvé & \textbf{singleton} \\
    \rowcolor{green!10}
    \textbf{NDCG@K} & Qualité du ranking pondérée par log-décrue & multi-GT, ordre fin \\
    M3 -- LLM-as-judge & Pertinence sémantique hors-GT & toutes -- \textbf{prochaine étape} \\
    \bottomrule
  \end{tabular}
  \end{center}

  \vspace{0.5em}
  \keybox{\textbf{Lecture cible} : reporter \emph{séparément} les scores sur le régime singleton
  et le régime multi-GT pour ne plus confondre « modèle parfait » avec « bench dégénéré ».}
\end{frame}

\begin{frame}{Détail des métriques (1/2) -- M1, M2, Hit@K}
  \footnotesize

  \textbf{\color{accent}M1 -- Recall@K \ (régime multi-GT)}
  \vspace{-0.2em}
  \[
    M1(q) = \frac{|\,\mathrm{GT}_q \cap R_q[{:}K]\,|}{|\,\mathrm{GT}_q\,|}
    \qquad
    \overline{M1} = \frac{1}{|Q|}\sum_q M1(q)
  \]
  \vspace{-0.6em} \\
  \textbf{Mesure} : fraction de la GT retrouvée dans le top-$K$ \ (range $[0,1]$). \\
  \textbf{Limite} : aveugle au rang ; dégénère en Hit@K quand $|\mathrm{GT}|{=}1$.

  \vspace{0.3em}\hrulefill\vspace{0.2em}

  \textbf{\color{accent}M2 -- Rang moyen normalisé \ (régime multi-GT, custom)}
  \vspace{-0.2em}
  \[
    M2(q) = \frac{x - (K{+}1)}{\frac{b'+1}{2} - (K{+}1)}
    \quad
    x = \tfrac{1}{|\mathrm{GT}|}\!\!\sum_{a\in\mathrm{GT}}\!\!\mathrm{rang}(a),\
    b' = \min(|\mathrm{GT}|, K)
  \]
  \vspace{-0.6em} \\
  \textbf{Mesure} : qualité du re-rank \emph{interne} à la GT trouvée \ (range $[0,1]$). \\
  \textbf{Limite} : couplée à M1 (dégénérée si $M1{=}0$), formulation non-standard.

  \vspace{0.3em}\hrulefill\vspace{0.2em}

  \textbf{\color{ok}Hit@K \ (régime singleton)}
  \vspace{-0.2em}
  \[
    \mathrm{Hit@K}(q) =
    \begin{cases}
      1 & \text{si } |\,\mathrm{GT}_q \cap R_q[{:}K]\,| \ge 1 \\
      0 & \text{sinon}
    \end{cases}
  \]
  \vspace{-0.6em} \\
  \textbf{Mesure} : présence d'au moins un bon résultat dans le top-$K$ (binaire par question). \\
  \textbf{Pertinence} : métrique naturelle du régime singleton (light user) ; sur strict on aura $\mathrm{Hit@K} \approx M1$ par construction.
\end{frame}

\begin{frame}{Détail des métriques (2/2) -- MRR@K, NDCG@K, M3}
  \footnotesize

  \textbf{\color{ok}MRR@K -- Mean Reciprocal Rank \ (régime singleton)}
  \vspace{-0.2em}
  \[
    \mathrm{RR}(q) =
    \begin{cases}
      \dfrac{1}{\mathrm{rang}(a^*)} & \text{si } \exists\, a^* \in \mathrm{GT}_q \cap R_q[{:}K] \\[0.3em]
      0 & \text{sinon}
    \end{cases}
    \quad a^* = \text{premier GT par rang}
  \]
  \vspace{-0.6em} \\
  \textbf{Mesure} : à quel rang apparaît le premier GT trouvé (rang 1 $\to$ 1 ; rang 2 $\to$ 0,5 ; rang 10 $\to$ 0,1). \\
  \textbf{Convention} : cappé à $K{=}10$ (rang $>K \to 0$) ; référence reco system pour le régime singleton.

  \vspace{0.3em}\hrulefill\vspace{0.2em}

  \textbf{\color{ok}NDCG@K -- Normalized Discounted Cumulative Gain \ (régime multi-GT)}
  \vspace{-0.2em}
  \[
    \mathrm{NDCG@K}(q) = \frac{\sum_{i=1}^{K} \mathrm{rel}_i \,/\, \log_2(i{+}1)}
                              {\sum_{i=1}^{\min(|\mathrm{GT}|,K)} 1 \,/\, \log_2(i{+}1)}
    \quad \mathrm{rel}_i =
    \begin{cases} 1 & R_q[i]\in\mathrm{GT}_q \\ 0 & \text{sinon} \end{cases}
  \]
  \vspace{-0.6em} \\
  \textbf{Mesure} : ranking pondéré par décrue logarithmique -- récompense couverture ET ordre. Hit en rang 1 vaut $1$, en rang 2 vaut $0{,}63$, en rang 10 vaut $0{,}29$. \\
  \textbf{Convention} : rel binaire pour l'instant ; multi-niveau après enrichissement GT (Chantier 3).

  \vspace{0.3em}\hrulefill\vspace{0.2em}

  \textbf{\color{crosscol}M3 -- LLM-as-judge \ (qualité hors-GT)}
  \vspace{-0.2em}
  \[
    M3(q) = \frac{2\cdot \#n_2 + \#n_1}{2K}
    \quad\text{avec } n_2 / n_1 / n_0 = \text{pertinent} \,/\, \text{proche} \,/\, \text{non pertinent}
  \]
  \vspace{-0.6em} \\
  \textbf{Mesure} : pertinence sémantique du top-$K$ indépendamment de la GT ; capte les vrais positifs absents de la GT. \\
  \textbf{Coût} : $|Q|\!\times\!K\approx 10\mathrm{k}$ appels Gemma 4 26B.
\end{frame}

\begin{frame}{Chantier 1 -- conventions retenues et plan d'impl.}
  \footnotesize
  \textbf{Conventions à valider} :
  \begin{itemize}\setlength{\itemsep}{0.1em}

\begin{frame}{M3 -- interprétation, enjeux, questions à Johnny}
  \footnotesize

  \textbf{Pourquoi B2-a cosine bat les méthodes graphe/fusion sur M3 ?}
  \begin{itemize}\setlength{\itemsep}{0.05em}
    \item Les méthodes graphe (B3-e, PPR, B4-e RRF) tirent des voisins par citation
    \emph{pour gonfler le recall-GT}. Ces voisins augmentent la couverture mais
    \textbf{introduisent des items moins pertinents sémantiquement}.
    \item PPR $\alpha=0{,}95$ (diffusion forte) maximise M1 étendu (0,440) au prix
    d'un top-K saturé de noeuds populaires \emph{peu pertinents pour la question}
    (M3 = 0,214, plancher).
    \item \textbf{Optimiser le recall-GT ne gagne pas la pertinence.} M3 mesure ce
    que M1 ne voit pas : la précision sémantique hors-GT.
  \end{itemize}

  \vspace{0.4em}
  \textbf{Côté JP : B4-f remonte (0,154 $\to$ 0,424) -- pourquoi ?}
  \begin{itemize}\setlength{\itemsep}{0.05em}
    \item B4-f cite-pondéré ramène les \emph{grands arrêts populaires} sur le thème.
    Pas les pourvois exacts de la GT (d'où le M1 désastreux) mais des JP
    \textbf{réellement pertinentes au juge LLM}.
    \item C'est \textbf{exactement le biais GT-sparse que M3 doit corriger} :
    quand la GT JP = pourvois CC précis, on rate la pertinence sémantique.
  \end{itemize}

  \vspace{0.4em}
  \begin{columns}[T]
    \column{0.55\textwidth}
    \warnbox{\scriptsize \textbf{Question à Johnny -- dénominateur M3.}
    Actuellement \texttt{k\_fixed} ($2K{=}20$ items théoriques). Alternative :
    \texttt{k\_ranking} (taille réelle du ranking). N'impacte que B3-e (~9,86) et
    B4-d (~9,57). \textbf{Switch = re-agrégation seule, zéro re-jugement} : choix
    à trancher avant figeage des chiffres.}

    \column{0.43\textwidth}
    \keybox{\scriptsize \textbf{Implications stratégiques :}
    \begin{enumerate}\setlength{\itemsep}{0.05em}
      \item Cosine pur n'est pas qu'un baseline -- il est \textbf{champion M3}
      des deux côtés.
      \item B3-e domine M1 strict mais \textbf{n'est pas mieux que B2-a sur M3}.

    \end{enumerate}}

\begin{frame}{Chantier 3 -- pivot conceptuel}
  \footnotesize
  \textbf{Plutôt que} :
  \vspace{0.2em}
  \begin{center}
  \emph{« GT extraite du texte de la question »}\\
  $\to$ singletons mécaniques, métriques range dégénérées
  \end{center}

  \vspace{0.5em}
  \textbf{Adopter} :
  \begin{center}
  \emph{« GT ancrée dans des décisions réelles »}\\
  $\to$ visa du juge $\Rightarrow$ 3-8 articles par question naturellement
  \end{center}

  \vspace{0.8em}

  Une métrique différente ne suffit pas -- il faut une GT structurellement plus riche.
  \textbf{Reverse-engineering :} pour un arrêt CA/Cass dont on connaît le visa, formuler
  la question « quels articles s'appliquent à [résumé des faits] ? ».
  GT = visa de l'arrêt. Scalable à milliers de questions, gold humain.}

  \vspace{0.5em}
  \textbf{À arbitrer avec Johnny} : valider l'investissement reverse-engineering vs purs
  enrichissements gratuits (visa + Légifrance).
\end{frame}

\section{Chantier 4 -- LightGCN (résultats sur G0 brut)}

\begin{frame}{LightGCN -- en une diapo}
  \footnotesize
  \okbox{\textbf{LightGCN bat PPR sur 2 régimes sur 3} -- sur le graphe \textbf{G0 brut},
  sans nettoyage. \textbf{Critère du handoff atteint $\times 2$.}}

  \vspace{0.4em}
  \begin{itemize}\setlength{\itemsep}{0.15em}
    \item \textbf{Design A} : on propage les embeddings BGE-M3 sur le graphe de citations
    Art$\leftrightarrow$JP ; les questions sont des « users » BGE-M3 \emph{figés},
    scorés en \textbf{cosinus}.
    \item \textcolor{ok}{\textbf{$\uparrow$}}~\textbf{Articles strict} : bat PPR
    (NDCG $0{,}533$ vs $0{,}334$), \textbf{égale} le champion B3-e ($0{,}711$) et gagne
    NDCG/MRR.
    \item \textcolor{ok}{\textbf{$\uparrow$}}~\textbf{JP} : \textbf{nouveau champion}
    toutes méthodes (M1 $0{,}437 >$ B4-e $0{,}416 >$ PPR $0{,}407$).
    \item PPR garde l'avantage en \textbf{articles étendu} (diffusion large $\to$ rappel).
  \end{itemize}

  \vspace{0.4em}
  \keybox{\scriptsize \textbf{Caveat} : la victoire vs PPR (untrained $K{=}2$) est
  \textbf{déterministe} (aucun apprentissage). Le gain de l'entraînement
  est \textbf{validé multi-seed} (3 seeds) : M1 strict $\mathbf{0{,}704 \pm 0{,}007}$,
  $\Delta$ vs untrained $\mathbf{+0{,}051}$ (toujours positif, $\in [+0{,}045 ; +0{,}059]$).
  Hyperparams $(\tau, \lambda, \mathrm{lr})$ non encore tunés.}
\end{frame}

\begin{frame}{LightGCN -- décomposition de l'ablation (M1 strict articles)}
  \footnotesize
  L'ablation $K{=}0$ ($\equiv$ cosine BGE) sépare proprement chaque contribution :

  \vspace{0.5em}
  \begin{center}\scriptsize
  \renewcommand{\arraystretch}{1.5}
  \setlength{\tabcolsep}{8pt}
  \begin{tabular}{@{}l c l@{}}
    \toprule
    \textbf{Méthode} & \textbf{M1 strict} & \textbf{Ce que ça isole} \\

    Texte seul (cosine BGE-M3) & $0{,}459$ & le sémantique pur \\

    & \textcolor{ok}{\textbf{$+0{,}194$ = apport du GRAPHE}} \\
    \;\;\;\; + apprentissage (BPR cosinus, \emph{3 seeds}) & $\mathbf{0{,}704 \pm 0{,}007}$

    \bottomrule
  \end{tabular}

  \vspace{0.6em}
  \keybox{Chaque étage \textbf{ajoute}. Le graphe de citations apporte massivement au

  (3 seeds : $0{,}698 / 0{,}712 / 0{,}701$, $\Delta \in [+0{,}045 ; +0{,}059]$ toujours positif).}

  \vspace{0.4em}
  \textbf{Over-smoothing} visible : $K_0 < K_1 < \mathbf{K_2} > K_3$.
  $K{=}2$ optimal $\Rightarrow$ conforme à la théorie LightGCN.
\end{frame}

\begin{frame}{LightGCN -- trois leçons techniques}
  \footnotesize
  \begin{enumerate}\setlength{\itemsep}{0.5em}
    \item \textbf{Scoring cosinus obligatoire.} En produit scalaire, la propagation
    \textbf{gonfle la norme des hubs} $\Rightarrow$ biais de popularité qui
    \textbf{effondre} strict + JP (M1 $\approx 0{,}02$).
    La L2-normalisation retire ce biais.
    \item \textbf{Entraînement BPR cosinus + ancrage anti-drift.}
    Loss = $\mathrm{BPR}_{\cos}(\tau{=}0{,}1)$ + $\lambda \lVert E_0 - \mathrm{BGE} \rVert^2$
    (ancrage à l'init BGE pour éviter le drift sur seulement 840 paires d'entraînement).
    Sans ces 2 corrections, l'entraînement naïf s'effondre.
    \item \textbf{Pas de PyG nécessaire.} Propagation pure PyTorch sur la matrice de citations
    sparse (CPU+MPS, $\sim 5$ min). L'écosystème \texttt{torch\_geometric} aurait été du surplus.
  \end{enumerate}

  \vspace{0.4em}
  \warnbox{\scriptsize \textbf{Tradeoff entraînement} : à $K{=}2$ fixé, l'entraînement
  \textbf{échange un peu de JP} ($0{,}437 \to 0{,}426$, perte $-0{,}011$) \textbf{contre
  beaucoup de précision articles} ($0{,}653 \to 0{,}705$, gain $+0{,}052$).
  Les 840 paires de train sont des positifs \emph{articles} -- \textbf{zéro supervision JP}.
  D'où le choix : \emph{trained} pour les articles, \emph{untrained} pour le JP.}
\end{frame}

\begin{frame}{LightGCN -- limites honnêtes \& suite}
  \footnotesize
  \warnbox{\scriptsize Tout ceci sur le graphe \textbf{G0 brut} : 36~\% d'articles résolus,
  48~\% de nœuds morts, graphe binaire. Le nettoyage $G_0 \to V_n$ est de
  \textbf{l'upside non exploité}.}

  \vspace{0.4em}
  \textbf{Autres limites} :
  \begin{itemize}\setlength{\itemsep}{0.05em}
    \item $840$ paires d'entraînement seulement (732 q hors cohorte) $\to$ apprentissage = calibration, pas un gros gain.
    \item \textbf{JP non supervisé} (zéro paire positive côté JP).
    \item PPR garde l'avantage en étendu (diffusion plus large).
  \end{itemize}

  \vspace{0.4em}
  \textbf{Prochaines étapes priorisées} :
  \begin{enumerate}\setlength{\itemsep}{0.1em}
    \item \textcolor{ok}{\textbf{$\checkmark$ Multi-seed validé}} (3 seeds, $\pm 0{,}007$). Reste : ablation init (zéro vs aléatoire) pour décomposer le $+0{,}051$.
    \item \textbf{Régime D} : injecter la supervision JP$+$article (cf. ADR-001).
    \item $G_0 \to V_1$ : retirer les 41\,824 nœuds morts, re-mesurer.
    \item Combler l'\textbf{étendu} (variante C : questions dans le graphe).
    \item Tuning $(\tau, \lambda, \mathrm{lr})$ + grille $K$ autour de 2.
  \end{enumerate}
\end{frame}
