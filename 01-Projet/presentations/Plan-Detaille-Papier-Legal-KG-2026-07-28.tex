\begin{frame}{Pourquoi ce problème est-il important ?}
  \papersection{1. Introduction -- Motivation}
  \footnotesize
  \textbf{Une question juridique appelle deux lectures complémentaires du droit.}

  \vspace{0.8em}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.46\textwidth}
      \textbf{\color{accent}Cadre normatif : les articles}
      \begin{itemize}\setlength{\itemsep}{0.38em}
        \item Ils définissent les règles applicables.
        \item Ils permettent d'identifier le premier cadre de réponse.
      \end{itemize}
    \end{column}
    \begin{column}{0.46\textwidth}
      \textbf{\color{ok}Application : les jurisprudences}
      \begin{itemize}\setlength{\itemsep}{0.38em}
        \item Elles montrent comment les règles sont interprétées.
        \item Elles permettent de soutenir ou de contester un argument précis.
      \end{itemize}
    \end{column}
  \end{columns}

  \vspace{1.0em}
  \keybox{\scriptsize Le besoin central n'est pas seulement de trouver un texte proche,
  mais de retrouver les sources normatives et jurisprudentielles pertinentes pour la question.}
\end{frame}

        \item Epoch et paramètres sélectionnés sur les folds.
        \item Replay final sans early stopping sur l'eval.
      \end{itemize}
    \end{column}
  \end{columns}

  \vspace{0.9em}
